# Zeroize GRUB theme - fallback variant
#
# Identical to theme.txt except that it names no font anywhere, so GRUB uses
# whichever face it already loaded (unicode.pf2, loaded by config.cfg before
# this is reached).
#
# This exists because a GRUB theme that references a font GRUB has not loaded
# is discarded WHOLE, with no diagnostic of any kind: the menu simply appears
# as plain text on black, which looks exactly like a theme that was never
# configured. Staking the entire branded menu on a custom font loading turned
# out to be a bad bet - the fonts are cross-compiled from a host running a
# newer GRUB than the image, and a PF2 the bootloader will not read is
# indistinguishable, from the build side, from one it will.
#
# So the branding degrades instead of vanishing. The background, the layout,
# the spacing and the colours all survive here; only the typeface is lost.
#
# Sizes are omitted along with the font names: an item_height tuned for 22px
# Inter is wrong for 16px Unifont, and GRUB sizes rows from the font when it
# is not told otherwise.

title-text: ""
desktop-image: "background.png"
desktop-color: "#16181D"

# Full-screen for the same reason as the main theme: see its header.
terminal-left: "0"
terminal-top: "0"
terminal-width: "100%"
terminal-height: "100%"
terminal-border: "0"
